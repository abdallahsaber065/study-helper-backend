157 @validator("secret_key") 

158 def validate_secret_key(cls, v: str) -> str: 

159 """Ensure secret key is sufficiently long.""" 

160 if len(v) < 32: 

161 raise ValueError("SECRET_KEY must be at least 32 characters long") 

162 return v 

163 

164 @validator("database_url") 

165 def validate_database_url(cls, v: str) -> str: 

166 """Validate database URL format.""" 

167 if not v.startswith(("postgresql", "sqlite")): 

168 raise ValueError( 

169 "DATABASE_URL must be a PostgreSQL or SQLite connection string" 

170 ) 

171 return v 

172 

173 @validator("environment") 

174 def validate_environment(cls, v: str) -> str: 

175 """Ensure environment is valid.""" 

176 if v not in ["development", "testing", "staging", "production"]: 

177 raise ValueError( 

178 "ENVIRONMENT must be one of: development, testing, staging, production" 

179 ) 

180 return v 

181 

182 @validator("log_level") 

183 def validate_log_level(cls, v: str) -> str: 

184 """Ensure log level is valid.""" 

185 if v.upper() not in ["DEBUG", "INFO", "WARNING", "ERROR", "CRITICAL"]: 

186 raise ValueError( 

187 "LOG_LEVEL must be one of: DEBUG, INFO, WARNING, ERROR, CRITICAL" 

188 ) 

189 return v.upper() 

190 

191 @validator("log_format") 

192 def validate_log_format(cls, v: str) -> str: 

193 """Ensure log format is valid.""" 

194 if v not in ["json", "console"]: 

195 raise ValueError("LOG_FORMAT must be either 'json' or 'console'") 

196 return v
